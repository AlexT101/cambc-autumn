    debug_block({
        const bool enemy = enemies && safety.bound(p, world.me().taken).threat;
        bool camp = false;
        for (const int g : world.map().groups_near(p, world.map().max_camp_reach)) {
            const auto& members = world.map().groups[g].members;
            for (std::size_t i = 0; i < members.size() && !camp; ++i) {
                if (members[i].strength <= 0 || dist(p, members[i].pos) > members[i].reach) continue;
                const auto& c = world.camps[g];
                camp = (c.state == CampState::respawning && c.respawn_round <= world.round + 1)
                       || (c.state == CampState::alive && c.health[i] > 0);
            }
            if (camp) break;
        }
        if (enemy != risk.enemy || (!enemy && !camp) != risk.safe())
            std::printf("RISK-ORACLE-MISMATCH %d %d %d\n", world.round, p.x, p.y);
        static unsigned checks = 0;
        if (++checks % 1024 == 0) std::printf("RISK-ORACLE-CHECKS %u\n", checks);
    });
